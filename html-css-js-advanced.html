⚡ 1. ES6+ 現代 JavaScript 語法

箭頭函數（Arrow Function）

// 傳統函數
function add(a, b) { return a + b; }

// 箭頭函數（等同上方）
const add = (a, b) => a + b;

// 單一參數可省略括號
const double = x => x * 2;

// 陣列方法常用箭頭函數
const altitudes = treks.map(t => t.altitude);
const highTreks = treks.filter(t => t.altitude > 4000);

解構賦值（Destructuring）

// 物件解構
const trek = { name: '南崎巴扎', altitude: 3440, day: 4 };
const { name, altitude, day } = trek;  // 直接取出屬性

// 陣列解構
const [first, second, ...rest] = ['加德滿都', '盧卡拉', '法克定', '南崎巴扎'];
// first = '加德滿都', second = '盧卡拉', rest = ['法克定', '南崎巴扎']

// 函數參數中解構
const showTrek = ({ name, altitude }) => `${name}（${altitude}m）`;

展開運算子（Spread / Rest）

// 展開陣列
const arr1 = [1, 2, 3];
const arr2 = [...arr1, 4, 5];   // [1, 2, 3, 4, 5]

// 複製物件並修改屬性
const original = { name: '南崎巴扎', altitude: 3440 };
const updated  = { ...original, altitude: 3500 };  // 覆蓋 altitude

// Rest 參數（收集剩餘參數）
const sum = (...nums) => nums.reduce((a, b) => a + b, 0);

🏗️ 4. ES6 Class（類別）

class Hiker {
  constructor(name, age, level) {
    this.name  = name;
    this.age   = age;
    this.level = level;
    this.totalDistance = 0;
  }

  trek(km) {
    this.totalDistance += km;
    return `${this.name} 步行了 ${km}km，累計 ${this.totalDistance}km`;
  }

  get info() {  // Getter
    return `${this.name}（${this.age}歲，${this.level}）`;
  }
}

// 繼承（Inheritance）
class GuideHiker extends Hiker {
  constructor(name, age, guideYears) {
    super(name, age, '嚮導');    // 呼叫父類別 constructor
    this.guideYears = guideYears;
  }
  introduce() {
    return `我是嚮導 ${this.name}，有 ${this.guideYears} 年帶隊經驗`;
  }
}
